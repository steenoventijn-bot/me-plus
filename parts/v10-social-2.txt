// Keep message and image nodes alive across polling, including renewed signed photo URLs.
function v142SyncChatRows(box,rows,messages,makeRow,bindRow){
 const wanted=new Set(messages.map(m=>m.id));let changed=false;
 for(const [id,entry] of rows){if(!wanted.has(id)){entry.node.remove();rows.delete(id);changed=true}}
 let previous=null;
 for(const m of messages){
  const media=m.media_path||(m.mediaUrl||'').split('?')[0];
  const reply=messages.find(x=>x.id===m.reply_to_id);
  const key=JSON.stringify([m.body,m.message_type,m.battle_id,m.reply_to_id,reply?.body,media,!!m.read_at,m.reactions||[]]);
  let entry=rows.get(m.id);
  if(!entry||entry.key!==key){
   const node=makeRow(m),oldImage=entry?.node.querySelector('img'),newImage=node.querySelector('img');
   if(oldImage&&newImage&&entry.media===media)newImage.replaceWith(oldImage);
   if(entry)entry.node.replaceWith(node);
   entry={node,key,media};rows.set(m.id,entry);bindRow(node);changed=true;
  }
  const next=previous?previous.nextElementSibling:box.firstElementChild;
  if(next!==entry.node){box.insertBefore(entry.node,next);changed=true}
  previous=entry.node;
 }
 return changed;
}
