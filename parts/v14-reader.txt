/* Reading rewards are decided by one server-side, four-option quiz. */
const K14_API='https://npolnflektojripucyvq.supabase.co/functions/v1/me-plus-knowledge-v14-api';
let k14Reader=null,k14Sequence=0;
function k14Seconds(a){const words=['intro','body','extra_fact','why_it_matters'].map(k=>String(a?.[k]||'').trim()).filter(Boolean).join(' ').split(/\s+/u).filter(Boolean).length;return Math.max(1,Math.ceil(words*.3))}
readLabel=function(a){const seconds=k14Seconds(a);return seconds<60?`± ${seconds} sec lezen`:`± ${Math.ceil(seconds/60)} min lezen`};
function k14Api(op,data){return call(K14_API,op,data)}
function k14Status(){const s=k14Reader,t=el('k14-status');if(!s||!t)return;if(s.attempt){t.textContent=s.attempt.correct?(s.attempt.awarded?'Afgerond · 1 punt verdiend':'Afgerond · al eerder beloond'):'Afgerond · 0 punten';return}if(s.error){t.textContent=s.error;return}const remaining=Math.max(0,s.min-s.active);t.textContent=remaining?`${fmtSeconds(remaining)} actief lezen tot je quizvraag`:'Tekst uitgelezen · je quizvraag staat klaar';el('k14-quiz-open').disabled=remaining>0}
async function k14Tick(){const s=k14Reader;if(!s?.sessionId||s.busy||s.attempt||s.active>=s.min||document.hidden)return;s.busy=true;try{const r=await k14Api('tick',{sessionId:s.sessionId});if(k14Reader!==s)return;s.active=r.activeSeconds;s.min=r.minSeconds;s.error='';k14Status();if(r.done)clearInterval(s.timer)}catch(e){if(k14Reader===s){s.error='Geen verbinding. Je leestijd wordt hervat zodra je weer online bent.';k14Status()}}finally{s.busy=false}}
openArticle=async function(a){if(!a)return;closeArticle(false);const seq=++k14Sequence,page=document.createElement('div');page.className='article-page v9-article k14-article';page.id='article-page';page.innerHTML=`<div class="article-inner"><button type="button" class="icon-btn article-close" id="article-close" aria-label="Sluiten">${svg('close')}</button><div class="k14-kicker">${esc(a.category)} <span>· ${readLabel(a)}</span></div><h1>${esc(a.title)}</h1><div class="article-lead">${esc(a.intro)}</div><div class="article-save"><button class="button line small" id="article-save">${server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'}</button></div><div class="article-body"><p>${esc(a.body)}</p>${a.extra_fact?`<h3>Extra feit</h3><p>${esc(a.extra_fact)}</p>`:''}${a.why_it_matters?`<h3>Waarom dit interessant is</h3><p>${esc(a.why_it_matters)}</p>`:''}</div><section class="k14-quiz" id="k14-quiz"><span class="k14-eyebrow">EVEN CHECKEN</span><h2>Wat neem je mee?</h2><p>Eén vraag, vier antwoorden. Goed is 1 punt, fout is 0. Je eerste antwoord telt.</p><p id="k14-status" role="status">Leessessie starten…</p><button id="k14-quiz-open" class="button primary" disabled>Beantwoord de vraag</button></section><p class="k14-time-note">Leestijd is een schatting op basis van 200 woorden per minuut.</p></div>`;document.body.appendChild(page);document.body.classList.add('v9-reading');el('article-close').onclick=()=>closeArticle();el('article-save').onclick=async()=>{try{await toggleSave(a.id);el('article-save').textContent=server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'}catch(e){toast(e.message)}};const start=async()=>{try{const r=await k14Api('start',{articleId:a.id});if(seq!==k14Sequence)return;k14Reader={article:a,sessionId:r.sessionId,min:r.minSeconds,active:r.activeSeconds,attempt:r.attempt,busy:false,error:'',timer:null};k14Status();if(r.attempt){el('k14-quiz-open').remove();return}el('k14-quiz-open').textContent='Beantwoord de vraag';el('k14-quiz-open').onclick=k14OpenQuestion;k14Reader.timer=setInterval(k14Tick,4000)}catch(e){if(seq!==k14Sequence)return;el('k14-status').textContent='Leessessie kon niet starten. Je kunt de tekst alvast lezen.';const btn=el('k14-quiz-open');btn.disabled=false;btn.textContent='Opnieuw verbinden';btn.onclick=start}};await start()};
async function k14OpenQuestion(){const s=k14Reader,btn=el('k14-quiz-open');if(!s||s.active<s.min)return;btn.disabled=true;try{const q=await k14Api('question',{sessionId:s.sessionId});if(k14Reader!==s)return;const box=el('k14-quiz');box.innerHTML=`<span class="k14-eyebrow">KENNISQUIZ · 1 PUNT</span><h2>${esc(q.question)}</h2><p>Er is één goed antwoord. Je eerste antwoord telt.</p><div class="k14-options" role="group" aria-label="Antwoordopties">${q.options.map((option,i)=>`<button data-answer="${i}"><b>${'ABCD'[i]}</b><span>${esc(option)}</span></button>`).join('')}</div><div id="k14-feedback" aria-live="polite"></div>`;box.querySelectorAll('[data-answer]').forEach(b=>b.onclick=()=>k14Answer(Number(b.dataset.answer)));box.scrollIntoView({behavior:matchMedia('(prefers-reduced-motion: reduce)').matches?'auto':'smooth',block:'center'})}catch(e){btn.disabled=false;toast(e.message)}}
async function k14Answer(choice){const s=k14Reader;if(!s||s.answering||s.attempt)return;s.answering=true;const buttons=[...document.querySelectorAll('[data-answer]')];buttons.forEach(b=>b.disabled=true);try{const r=await k14Api('answer',{sessionId:s.sessionId,choice,localDate:localISO()});if(k14Reader!==s)return;s.attempt=r;buttons.forEach(b=>{const n=Number(b.dataset.answer);b.classList.toggle('correct',n===r.correctIndex);b.classList.toggle('incorrect',n===r.selectedIndex&&!r.correct);if(n===r.correctIndex)b.setAttribute('aria-label',b.textContent+' — goed antwoord')});el('k14-feedback').innerHTML=`<h3>${r.correct?(r.awarded?'Goed! +1 punt':'Goed! Dit artikel was al beloond.'):'Nog niet goed · 0 punten'}</h3><p>${esc(r.explanation)}</p><button id="k14-done" class="button line">Terug naar kennis</button>`;el('k14-done').onclick=()=>closeArticle();if(r.newPoint){premiumPopV8('+1 kennispunt');await refreshBase()}}catch(e){if(k14Reader===s){el('k14-feedback').textContent=e.message+' Je kunt opnieuw verzenden.';buttons.forEach(b=>b.disabled=false)}}finally{s.answering=false}}
closeArticle=function(redraw=true){k14Sequence++;if(k14Reader?.timer)clearInterval(k14Reader.timer);k14Reader=null;v9StopReader();readSession=null;document.querySelectorAll('#article-page').forEach(x=>x.remove());document.body.classList.remove('v9-reading');if(redraw)render()};
