function reminderOffset(v){return Number(v==='at'?0:v||0)||0}
function nextOccurrence(e){if(!e.start||!e.date)return null;let d=new Date(`${e.date}T${e.start}:00`);const now=Date.now();if(d.getTime()>=now)return d;if(!e.repeat||e.repeat==='none')return d;let guard=0;while(d.getTime()<now&&guard++<500){if(e.repeat==='daily')d.setDate(d.getDate()+1);else if(e.repeat==='weekly')d.setDate(d.getDate()+7);else if(e.repeat==='monthly')d.setMonth(d.getMonth()+1);else break}return d}
