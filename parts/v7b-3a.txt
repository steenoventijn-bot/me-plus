async function syncReminderForEvent(e){const key=`event:${e.id}`;if(!e.start||e.reminder==='none'){try{await api7('cancel_reminder',{eventKey:key})}catch{}return}const occ=nextOccurrence(e);if(!occ)return;const scheduled=new Date(occ.getTime()-reminderOffset(e.reminder)*60000);if(scheduled.getTime()<Date.now()-120000)return;try{await api7('schedule_reminder',{eventKey:key,title:e.title,kind:e.type==='growth'?'growth':'appointment',repeat:e.repeat||'none',timezone:tz(),scheduledAt:scheduled.toISOString(),leadMinutes:reminderOffset(e.reminder)})}catch(err){console.warn('reminder',err)}}
async function syncAllReminders(){for(const e of state.events)await syncReminderForEvent(e)}

function fileData(f){return new Promise((res,rej)=>{const r=new FileReader();r.onload=()=>res(r.result);r.onerror=rej;r.readAsDataURL(f)})}
function pointPop(t){const x=document.createElement('div');x.className='point-pop';x.textContent=t;document.body.appendChild(x);setTimeout(()=>x.remove(),1000)}

async function toggleSave(id){try{const r=await api7('toggle_save',{articleId:id});r.saved?server.saved.add(id):server.saved.delete(id);server.extras.saved=server.saved.size;render()}catch(e){toast(e.message)}}
async function openArticle(a){if(!a)return;clearInterval(readTimer);let start={alreadyRead:true,minSeconds:0,sessionId:null,activeSeconds:0};try{start=await api7('start_read_v7',{articleId:a.id})}catch(e){console.warn(e)}readSession={article:a,sessionId:start.sessionId,min:start.minSeconds||0,active:start.activeSeconds||0,alreadyRead:start.alreadyRead,complete:false};const page=document.createElement('div');page.className='article-page';page.id='article-page';page.innerHTML=`<div class="article-inner"><button class="icon-btn article-close" id="article-close">${svg('close')}</button><div class="article-cat">${esc(a.category)} · ${readLabel(a)}</div><h1>${esc(a.title)}</h1><div class="article-lead">${esc(a.intro)}</div><div class="article-save"><button class="button line small" id="article-save">${server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'}</button></div><div class="article-body"><p>${esc(a.body)}</p>${a.extra_fact?`<h3>Extra feit</h3><p>${esc(a.extra_fact)}</p>`:''}${a.why_it_matters?`<h3>Waarom dit interessant is</h3><p>${esc(a.why_it_matters)}</p>`:''}</div></div>${start.alreadyRead?'<div class="timer done"><span><strong>✓</strong>gelezen</span></div>':`<div class="timer" id="read-timer"><span><strong>${fmtSeconds(start.minSeconds||0)}</strong>actief lezen</span></div>`}`;document.body.appendChild(page);el('article-close').onclick=()=>closeArticle();el('article-save').onclick=async()=>{await toggleSave(a.id);el('article-save').textContent=server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'};if(!start.alreadyRead&&start.sessionId){readTimer=setInterval(readTick,3000)}}
function fmtSeconds(s){return s>=60?`${Math.floor(s/60)}:${String(s%60).padStart(2,'0')}`:`0:${String(s).padStart(2,'0')}`}
async function readTick(){if(!readSession?.sessionId||document.hidden)return;try{const r=await api7('read_tick_v7',{sessionId:readSession.sessionId});readSession.active=r.activeSeconds;const remain=Math.max(0,r.minSeconds-r.activeSeconds),t=el('read-timer');if(t)t.innerHTML=`<span><strong>${fmtSeconds(remain)}</strong>${remain?'actief lezen':'klaar'}</span>`;if(r.done)await finishRead()}catch(e){console.warn(e)}}
async function finishRead(){if(readSession?.complete||!readSession?.sessionId)return;readSession.complete=true;clearInterval(readTimer);try{const r=await api7('complete_read_v7',{sessionId:readSession.sessionId,localDate:localISO()});const t=el('read-timer');if(t){t.classList.add('done');t.innerHTML='<span><strong>✓</strong>gelezen</span>'}if(r.firstRead){pointPop('+1 kennispunt');await refreshBase()} }catch(e){readSession.complete=false}}
function closeArticle(){clearInterval(readTimer);readTimer=null;readSession=null;el('article-page')?.remove();render()}
