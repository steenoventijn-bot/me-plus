 const pushOn=Notification.permission==='granted'&&!!server.push?.enabled;

function paintReadTimer(){
 const t=el('read-timer');if(!t||!readSession)return;
 const remain=Math.max(0,Math.ceil(readSession.visualRemaining??Math.max(0,(readSession.min||0)-(readSession.active||0))));
 t.innerHTML=`<span><strong>${fmtSeconds(remain)}</strong>${remain?'actief lezen':'controleren…'}</span>`;
}

async function syncReadProgressV7c(){
 if(!readSession?.sessionId||readSession.syncing||document.hidden)return;
 readSession.syncing=true;
 try{
   const r=await api7('read_tick_v7',{sessionId:readSession.sessionId});
   readSession.active=r.activeSeconds;
   readSession.min=r.minSeconds;
   readSession.visualRemaining=Math.max(0,r.minSeconds-r.activeSeconds);
   paintReadTimer();
   if(r.done)await finishRead();
 }catch(e){console.warn(e)}finally{if(readSession)readSession.syncing=false}
}

async function openArticle(a){
 if(!a)return;
 clearInterval(readTimer);clearInterval(readSyncTimer);readTimer=null;readSyncTimer=null;
 let start={alreadyRead:true,minSeconds:0,sessionId:null,activeSeconds:0};
 try{start=await api7('start_read_v7',{articleId:a.id})}catch(e){console.warn(e)}
 readSession={article:a,sessionId:start.sessionId,min:start.minSeconds||0,active:start.activeSeconds||0,visualRemaining:Math.max(0,(start.minSeconds||0)-(start.activeSeconds||0)),alreadyRead:start.alreadyRead,complete:false,syncing:false};
 const page=document.createElement('div');page.className='article-page';page.id='article-page';
 page.innerHTML=`<div class="article-inner"><button class="icon-btn article-close" id="article-close">${svg('close')}</button><div class="article-cat">${esc(a.category)} · ${readLabel(a)}</div><h1>${esc(a.title)}</h1><div class="article-lead">${esc(a.intro)}</div><div class="article-save"><button class="button line small" id="article-save">${server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'}</button></div><div class="article-body"><p>${esc(a.body)}</p>${a.extra_fact?`<h3>Extra feit</h3><p>${esc(a.extra_fact)}</p>`:''}${a.why_it_matters?`<h3>Waarom dit interessant is</h3><p>${esc(a.why_it_matters)}</p>`:''}</div></div>${start.alreadyRead?'<div class="timer done"><span><strong>✓</strong>gelezen</span></div>':`<div class="timer" id="read-timer"><span><strong>${fmtSeconds(readSession.visualRemaining)}</strong>actief lezen</span></div>`}`;
 document.body.appendChild(page);
 el('article-close').onclick=()=>closeArticle();
 el('article-save').onclick=async()=>{await toggleSave(a.id);el('article-save').textContent=server.saved.has(a.id)?'✓ Bewaard':'Bewaar voor later'};
 if(!start.alreadyRead&&start.sessionId){
   readTimer=setInterval(()=>{if(document.hidden||!readSession||readSession.complete)return;if(readSession.visualRemaining>0){readSession.visualRemaining=Math.max(0,readSession.visualRemaining-1);paintReadTimer()}else{syncReadProgressV7c()}},1000);
   readSyncTimer=setInterval(syncReadProgressV7c,5000);
 }
}

async function readTick(){return syncReadProgressV7c()}

async function finishRead(){
 if(readSession?.complete||!readSession?.sessionId)return;
 readSession.complete=true;clearInterval(readTimer);clearInterval(readSyncTimer);readTimer=null;readSyncTimer=null;
 try{
   const r=await api7('complete_read_v7',{sessionId:readSession.sessionId,localDate:localISO()});
   const t=el('read-timer');if(t){t.classList.add('done');t.innerHTML='<span><strong>✓</strong>gelezen</span>'}
   if(r.firstRead){pointPop('+1 kennispunt');await refreshBase()}
 }catch(e){
   readSession.complete=false;
   if(e?.data?.remaining!=null)readSession.visualRemaining=Math.max(0,Number(e.data.remaining)||0);
   paintReadTimer();
   if(!readTimer)readTimer=setInterval(()=>{if(document.hidden||!readSession||readSession.complete)return;if(readSession.visualRemaining>0){readSession.visualRemaining=Math.max(0,readSession.visualRemaining-1);paintReadTimer()}else{syncReadProgressV7c()}},1000);
   if(!readSyncTimer)readSyncTimer=setInterval(syncReadProgressV7c,5000);
 }
}

function closeArticle(){clearInterval(readTimer);clearInterval(readSyncTimer);readTimer=null;readSyncTimer=null;readSession=null;el('article-page')?.remove();render()}

async function registerSW(){if(!('serviceWorker'in navigator))return;try{const reg=await navigator.serviceWorker.register('/sw.js?v=7c');reg.update().catch(()=>{})}catch(e){console.warn('sw',e)}}
